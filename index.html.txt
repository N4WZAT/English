        // LOADER
        window.addEventListener('load', () => {
            setTimeout(() => {
                document.getElementById('intro-loader').style.opacity = '0';
                setTimeout(() => { document.getElementById('intro-loader').style.display = 'none'; }, 600);

        // SESSION TIMER LOGIC
        let sessionSeconds = 0;
        let sessionInterval = null;
        let isSessionRunning = false;

        function toggleSession() {
            const btn = document.getElementById('session-toggle');
            if (isSessionRunning) {
                clearInterval(sessionInterval);
                btn.innerHTML = '<i class="fas fa-play"></i>';
            } else {
                sessionInterval = setInterval(() => {
                    sessionSeconds++;
                    updateSessionDisplay();
                }, 1000);
                btn.innerHTML = '<i class="fas fa-pause"></i>';
            }
            isSessionRunning = !isSessionRunning;
        }

        function resetSession() {
            clearInterval(sessionInterval);
            sessionSeconds = 0;
            isSessionRunning = false;
            updateSessionDisplay();
            document.getElementById('session-toggle').innerHTML = '<i class="fas fa-play"></i>';
        }

        function updateSessionDisplay() {
            const h = Math.floor(sessionSeconds / 3600);
            const m = Math.floor((sessionSeconds % 3600) / 60);
            const s = sessionSeconds % 60;
            document.getElementById('session-clock').innerText = 
                `${h.toString().padStart(2,'0')}:${m.toString().padStart(2,'0')}:${s.toString().padStart(2,'0')}`;
        }

        // PARALLAX
        document.addEventListener('mousemove', (e) => {
            const x = (window.innerWidth / 2 - e.pageX) / 40;
            const y = (window.innerHeight / 2 - e.pageY) / 40;
            document.querySelector('.parallax-bg').style.transform = `translate(${x}px, ${y}px)`;
        });

        // SOUND
        window.addEventListener('click', () => { 
            const a = document.getElementById('royal-audio'); 
            a.volume = 0.25;

        function openSection(cat) {
            document.getElementById('main-portal').style.display = 'none';
            document.getElementById('detail-view').style.display = 'block';
            document.getElementById('view-title').innerText = cat;
            const list = document.getElementById('items-list');
            list.innerHTML = '';
            categories[cat].forEach(item => {
                const done = localStorage.getItem(item) === 'true';
                const div = document.createElement('div');
                div.className = `task-item ${done ? 'done' : ''}`;
                div.innerHTML = `<span style="font-family:'Amiri'; font-size:1.4rem;">${item}</span><div class="check-box"></div>`;
                div.onclick = () => {
                    const s = div.classList.toggle('done');
                    localStorage.setItem(item, s);
                    if(s) { const t = document.getElementById('toast'); t.style.display='block'; setTimeout(()=>t.style.display='none', 2000); }
                };
                list.appendChild(div);
            });
        }

        function closeSection() { 
            document.getElementById('main-portal').style.display = 'block'; 
            document.getElementById('detail-view').style.display = 'none'; 
        }

        function toggleAch() {
            const modal = document.getElementById('ach-modal');
            const content = document.getElementById('ach-content');
            if(modal.style.display === 'block') {
                modal.style.display = 'none';
            } else {
                content.innerHTML = '';
                let found = false;
                Object.keys(categories).forEach(cat => {
                    const done = categories[cat].filter(i => localStorage.getItem(i) === 'true');
                    if(done.length > 0) {
                        found = true;
                        let html = `<div style="margin-bottom:30px; border-bottom:1px solid #333; padding-bottom:15px;">
                                    <h4 style="color:var(--accent-gold); font-family:'Cinzel'; margin-bottom:10px; font-size:1.4rem;">${cat}</h4>`;
                        done.forEach(i => html += `<span style="display:inline-block; margin:5px; padding:8px 15px; border:1px solid var(--accent-gold); font-family:'Amiri'; font-size:1.1rem; background:rgba(212,175,55,0.1);">${i}</span>`);
                        html += `</div>`;
                        content.innerHTML += html;
                    }
                });
                if(!found) content.innerHTML = '<p style="text-align:center; opacity:0.5; font-size:1.3rem;">No achievements recorded yet.</p>';
                modal.style.display = 'block';
            }
        }

        function updateTimer(){
            const target = new Date('June 12, 2026 00:00:00').getTime();
            const now = new Date().getTime();
            const diff = target - now;
            const d = Math.floor(diff / (1000 * 60 * 60 * 24));
            const h = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            const m = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
            const s = Math.floor((diff % (1000 * 60)) / 1000);
            document.getElementById('timer').innerText = `${d}D : ${h}H : ${m}M : ${s}S`;
        }
        setInterval(updateTimer, 1000); 
        updateTimer();
